    workspaceId:localStorage.getItem('anicca_workspace_active_v1'),workspaceTitle:document.querySelector('[data-testid="dialogue-workspace-bar"] strong')?.textContent,

    focus:{tag:focus?.tagName,id:focus?.id,label:focus?.getAttribute('aria-label'),text:focus?.textContent?.trim().slice(0,80)},
    visibleControls:Array.from(document.querySelectorAll('button,input,textarea,select,a')).filter(el=>{const r=el.getBoundingClientRect(),s=getComputedStyle(el);return r.width&&r.height&&s.visibility!=='hidden'&&s.display!=='none'&&r.bottom>0&&r.y<innerHeight;}).map(el=>({tag:el.tagName,label:el.getAttribute('aria-label')||el.textContent?.trim().slice(0,80),disabled:el.disabled,...rect(el)})),

});}
